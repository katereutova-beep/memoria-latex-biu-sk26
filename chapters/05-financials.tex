מספר ההזמנות החודשי מתואר בפונקציה סוג־הרוויה
\begin{equation}
  N(t) \;=\; N_0 + (N_\infty - N_0)\left(1 - e^{-t/\tau}\right), \qquad N_0=25,\; N_\infty=200,\; \tau=14.
\end{equation}

\subsection{נקודת איזון}
נקודת האיזון נגזרת מהנוסחה הקלאסית $Q^{*} = F/(p-v)$, כאשר $F$ העלויות הקבועות החודשיות, $p$ המחיר הממוצע להזמנה ו-$v$ העלות המשתנה הממוצעת. עם תרומה של \num{\MContribPerOrder} ש״ח להזמנה, נדרשות כ-\num{\MBreakEvenOrdersEarly} הזמנות בחודש כדי לכסות את העלויות בשנה הראשונה. המודל מראה רווח תפעולי חודשי חיובי מחודש \num{\MBreakEvenMonth}.

\begin{figure}[h]
\centering
\includegraphics[width=0.85\textwidth]{figures/breakeven.pdf}
\caption{נקודת האיזון: עקומת ההזמנות החודשיות חוצה את הרף $F/(p-v)$ בחודש \num{\MBreakEvenMonth}. הרף קופץ בחודש \num{\MHireThree} עקב גיוס עובד שלישי.}

\subsection{סימולציית \en{Monte Carlo}}
